# Deployment readiness: dashboard/server.py's own dependency guard
# (_DEPS_OK) treats these as optional so a plain desktop install without
# them still runs main.py fine — but on Render the dashboard IS the
# product, so they are required here, not optional. python-multipart is
# specifically what FastAPI's UploadFile needs for /api/upload.
fastapi
uvicorn[standard]
cryptography
python-multipart

# PostgreSQL persistent memory: memory/postgres_repo.py's only external
# dependency, optional at import time (guarded — see that module) so this
# still runs with no DATABASE_URL set at all, falling back to the local
# JSON file. [binary] ships a prebuilt libpq, no system Postgres client
# headers needed on Render's build image.
psycopg[binary]>=3.1,<4

# Device-local-time fix: zoneinfo (stdlib) needs an actual IANA tz database
# to resolve names like "Asia/Kathmandu" and their DST rules. Most desktop
# OSes and full Linux images ship one system-wide, but Windows never does
# and some minimal container base images don't either — tzdata is the
# official, portable, pip-installable copy of that same database (not a
# hardcoded timezone), so this is required, not optional, wherever the web
# login's timezone is used.
tzdata
